## LC 78 — Subsets C++ sketch

```cpp
void dfs(int i, vector<int>& cur, vector<int>& nums, vector<vector<int>>& out) {
    if (i == (int)nums.size()) { out.push_back(cur); return; }
    dfs(i + 1, cur, nums, out);
    cur.push_back(nums[i]);
    dfs(i + 1, cur, nums, out);
    cur.pop_back();
}
```
