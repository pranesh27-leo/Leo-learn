# Reference Solution (spoiler)

## Reference: HashMap (simplified chaining)

```cpp
#include <string>
#include <vector>
#include <functional>
using namespace std;

class HashMap {
    vector<vector<pair<string,int>>> buckets;
    size_t n = 0;
    static size_t idx(const string& k, size_t B) {
        return hash<string>{}(k) % B;
    }
    void rehash() {
        auto old = move(buckets);
        buckets.assign(old.size() * 2, {});
        n = 0;
        for (auto& chain : old)
            for (auto& [k,v] : chain) put(k, v);
    }
public:
    HashMap(size_t B = 16) : buckets(B) {}
    void put(const string& key, int value) {
        if (n + 1 > buckets.size() * 0.75) rehash();
        size_t i = idx(key, buckets.size());
        for (auto& [k,v] : buckets[i])
            if (k == key) { v = value; return; }
        buckets[i].push_back({key, value});
        ++n;
    }
    bool get(const string& key, int& out) const {
        for (auto& [k,v] : buckets[idx(key, buckets.size())])
            if (k == key) { out = v; return true; }
        return false;
    }
};
```

## Reference: MinHeap

```cpp
class MinHeap {
    vector<int> h;
    void up(int i) {
        while (i > 0) {
            int p = (i - 1) / 2;
            if (h[p] <= h[i]) break;
            swap(h[p], h[i]);
            i = p;
        }
    }
    void down(int i) {
        int sz = (int)h.size();
        while (true) {
            int l = 2*i+1, r = 2*i+2, s = i;
            if (l < sz && h[l] < h[s]) s = l;
            if (r < sz && h[r] < h[s]) s = r;
            if (s == i) break;
            swap(h[i], h[s]);
            i = s;
        }
    }
public:
    void push(int x) { h.push_back(x); up((int)h.size()-1); }
    int top() const { return h.front(); }
    void pop() { h[0] = h.back(); h.pop_back(); if (!h.empty()) down(0); }
    int size() const { return (int)h.size(); }
};
```

## Reference: Trie (insert + startsWith)

```cpp
struct TrieNode {
    TrieNode* child[26]{};
    bool end = false;
};

class Trie {
    TrieNode* root = new TrieNode();
public:
    void insert(const string& w) {
        TrieNode* cur = root;
        for (char ch : w) {
            int i = ch - 'a';
            if (!cur->child[i]) cur->child[i] = new TrieNode();
            cur = cur->child[i];
        }
        cur->end = true;
    }
    bool startsWith(const string& p) {
        TrieNode* cur = root;
        for (char ch : p) {
            int i = ch - 'a';
            if (!cur->child[i]) return false;
            cur = cur->child[i];
        }
        return true;
    }
};
```

## Reference: LRU Cache (STL allowed in integration)

For LC 146 production solution you may use:

```cpp
class LRUCache {
    int cap;
    list<pair<int,int>> order;
    unordered_map<int, list<pair<int,int>>::iterator> mp;
public:
    LRUCache(int c) : cap(c) {}
    int get(int key) {
        auto it = mp.find(key);
        if (it == mp.end()) return -1;
        order.splice(order.begin(), order, it->second);
        return it->second->second;
    }
    void put(int key, int value) {
        auto it = mp.find(key);
        if (it != mp.end()) {
            it->second->second = value;
            order.splice(order.begin(), order, it->second);
            return;
        }
        if ((int)order.size() == cap) {
            mp.erase(order.back().first);
            order.pop_back();
        }
        order.push_front({key, value});
        mp[key] = order.begin();
    }
};
```
